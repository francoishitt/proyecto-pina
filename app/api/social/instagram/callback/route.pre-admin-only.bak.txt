import {NextRequest,NextResponse} from "next/server";import {cookies} from "next/headers";import {prisma} from "@/lib/prisma";export async function GET(req:NextRequest){const base=process.env.NEXT_PUBLIC_SITE_URL!;const jar=await cookies();const q=req.nextUrl.searchParams;if(!q.get("state")||q.get("state")!==jar.get("ig_oauth_state")?.value)return NextResponse.redirect(new URL("/admin/videos?error=state",base));const code=(q.get("code")||"").replace(/#_$/,"");if(!code)return NextResponse.redirect(new URL("/admin/videos?error=instagram",base));const redirect=`${base}/api/social/instagram/callback`;const body=new URLSearchParams({client_id:process.env.INSTAGRAM_APP_ID||"",client_secret:process.env.INSTAGRAM_APP_SECRET||"",grant_type:"authorization_code",redirect_uri:redirect,code});const sr=await fetch("https://api.instagram.com/oauth/access_token",{method:"POST",headers:{"Content-Type":"application/x-www-form-urlencoded"},body});if(!sr.ok)return NextResponse.redirect(new URL("/admin/videos?error=instagram_token",base));const short=await sr.json();const lu=new URL("https://graph.instagram.com/access_token");lu.searchParams.set("grant_type","ig_exchange_token");lu.searchParams.set("client_secret",process.env.INSTAGRAM_APP_SECRET||"");lu.searchParams.set("access_token",short.access_token);const lr=await fetch(lu);const long=lr.ok?await lr.json():short;const token=long.access_token||short.access_token;const pu=new URL("https://graph.instagram.com/me");pu.searchParams.set("fields","id,username,name,profile_picture_url");pu.searchParams.set("access_token",token);const pr=await fetch(pu);const user=pr.ok?await pr.json():{};await prisma.conexionSocial.upsert({where:{plataforma:"INSTAGRAM"},update:{accessToken:token,tokenExpiresAt:new Date(Date.now()+(long.expires_in||5184000)*1000),externalUserId:String(user.id||short.user_id||""),username:user.username,displayName:user.name,profileUrl:user.username?`https://www.instagram.com/${user.username}/`:null,avatarUrl:user.profile_picture_url,scope:"instagram_business_basic"},create:{plataforma:"INSTAGRAM",accessToken:token,tokenExpiresAt:new Date(Date.now()+(long.expires_in||5184000)*1000),externalUserId:String(user.id||short.user_id||""),username:user.username,displayName:user.name,profileUrl:user.username?`https://www.instagram.com/${user.username}/`:null,avatarUrl:user.profile_picture_url,scope:"instagram_business_basic"}});jar.delete("ig_oauth_state");return NextResponse.redirect(new URL("/admin/videos?conectado=instagram",base))}
